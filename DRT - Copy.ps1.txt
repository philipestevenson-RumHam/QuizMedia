$script:allData = $null
$script:job = $null
$script:timer = New-Object System.Windows.Forms.Timer
$script:timer.Interval = 500

function Set-GridData {
    param($data)
    $script:allData = $data
    $dataGrid.DataSource = [System.Collections.ArrayList]$data

    if ($dataGrid.Columns["FullPath"]) { $dataGrid.Columns["FullPath"].Visible = $false }

    foreach ($row in $dataGrid.Rows) {
        if ($row.Cells["Flagged"].Value -eq $true) {
            $row.DefaultCellStyle.BackColor = [System.Drawing.Color]::MistyRose
        }
    }

    $flagged = @($data | Where-Object { $_.Flagged })
    $flaggedSizeMB = 0
    if ($flagged.Count -gt 0) {
        $flaggedSizeMB = ($flagged | Measure-Object -Property SizeKB -Sum).Sum / 1024
    }
    $lblSummary.Text = "Total files: $($data.Count)   |   Flagged for review: $($flagged.Count)   ($([math]::Round($flaggedSizeMB,1)) MB)"
}

$btnScan.Add_Click({
    if (-not (Test-Path $txtFolder.Text)) {
        [System.Windows.Forms.MessageBox]::Show("Please select a valid folder.")
        return
    }

    $dateFieldMap = @{
        "LastWriteTime (Modified)"  = "LastWriteTime"
        "CreationTime (Created)"    = "CreationTime"
        "LastAccessTime (Accessed)" = "LastAccessTime"
    }
    $dateField  = $dateFieldMap[$cmbDateField.SelectedItem.ToString()]
    $cutoffDate = $dtCutoff.Value
    $folderPath = $txtFolder.Text

    $btnScan.Enabled = $false
    $btnExportAll.Enabled = $false
    $btnExportFlagged.Enabled = $false
    $progressBar.Visible = $true
    $lblStatus.Text = "Scanning... this may take a while for large folders."

    $scriptBlock = {
        param($FolderPath, $DateField, $CutoffDate)
        Get-ChildItem -Path $FolderPath -Recurse -File -ErrorAction SilentlyContinue | ForEach-Object {
            $owner = try { (Get-Acl $_.FullName -ErrorAction Stop).Owner } catch { "N/A" }
            $dateVal = $_.$DateField
            [PSCustomObject]@{
                Name         = $_.Name
                Folder       = $_.DirectoryName
                Extension    = $_.Extension
                SizeKB       = [math]::Round($_.Length / 1KB, 1)
                Created      = $_.CreationTime
                LastModified = $_.LastWriteTime
                LastAccessed = $_.LastAccessTime
                Owner        = $owner
                AgeDays      = [math]::Round((New-TimeSpan -Start $dateVal -End (Get-Date)).TotalDays,0)
                Flagged      = [bool]($dateVal -lt $CutoffDate)
                FullPath     = $_.FullName
            }
        }
    }

    $script:job = Start-Job -ScriptBlock $scriptBlock -ArgumentList $folderPath, $dateField, $cutoffDate

    $script:timer.Remove_Tick($script:timerHandler) 2>$null

    $script:timerHandler = {
        if ($script:job.State -eq "Completed") {
            $script:timer.Stop()
            $result = @(Receive-Job -Job $script:job)
            Remove-Job -Job $script:job -Force
            $progressBar.Visible = $false
            $btnScan.Enabled = $true
            $btnExportAll.Enabled = $true
            $btnExportFlagged.Enabled = $true
            Set-GridData -data $result
            $lblStatus.Text = if ($result.Count -gt 0) { "Scan complete." } else { "No files found." }
        }
        elseif ($script:job.State -eq "Failed") {
            $script:timer.Stop()
            $errMsg = (Receive-Job -Job $script:job 2>&1 | Out-String)
            Remove-Job -Job $script:job -Force
            $progressBar.Visible = $false
            $btnScan.Enabled = $true
            $lblStatus.Text = "Scan failed."
            [System.Windows.Forms.MessageBox]::Show("Scan failed: $errMsg")
        }
    }
    $script:timer.Add_Tick($script:timerHandler)
    $script:timer.Start()
})

$btnExportAll.Add_Click({
    if (-not $script:allData -or $script:allData.Count -eq 0) {
        [System.Windows.Forms.MessageBox]::Show("No data to export. Run a scan first.")
        return
    }
    $saveDialog = New-Object System.Windows.Forms.SaveFileDialog
    $saveDialog.Filter = "CSV files (*.csv)|*.csv"
    $saveDialog.FileName = "FolderInventory_All.csv"
    if ($saveDialog.ShowDialog() -eq "OK") {
        $script:allData | Export-Csv -Path $saveDialog.FileName -NoTypeInformation
        [System.Windows.Forms.MessageBox]::Show("Exported $($script:allData.Count) rows.")
    }
})

$btnExportFlagged.Add_Click({
    if (-not $script:allData -or $script:allData.Count -eq 0) {
        [System.Windows.Forms.MessageBox]::Show("No data to export. Run a scan first.")
        return
    }
    $flagged = @($script:allData | Where-Object { $_.Flagged })
    if ($flagged.Count -eq 0) {
        [System.Windows.Forms.MessageBox]::Show("No flagged files to export.")
        return
    }
    $saveDialog = New-Object System.Windows.Forms.SaveFileDialog
    $saveDialog.Filter = "CSV files (*.csv)|*.csv"
    $saveDialog.FileName = "FolderInventory_Flagged.csv"
    if ($saveDialog.ShowDialog() -eq "OK") {
        $flagged | Export-Csv -Path $saveDialog.FileName -NoTypeInformation
        [System.Windows.Forms.MessageBox]::Show("Exported $($flagged.Count) flagged rows.")
    }
})
